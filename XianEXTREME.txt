# ================= LOGIN =================
function Login-KeyAuth {
    param ($LicenseKey)

    Write-Host "[!] Connecting..." -ForegroundColor Yellow

    $initBody = @{
        type    = "init"
        ver     = $VERSION
        name    = $APP_NAME
        ownerid = $OWNER_ID
    }

    try {
        $initResponse = Invoke-RestMethod `
            -Uri $KEYAUTH_API `
            -Method Post `
            -Body $initBody

        if ($initResponse.success -ne $true) {
            Write-Host "Init Failed" -ForegroundColor Red
            return $false
        }

        $sessionid = $initResponse.sessionid

        $hwid = (Get-CimInstance win32_computersystemproduct).UUID

        $loginBody = @{
            type      = "license"
            key       = $LicenseKey
            hwid      = $hwid
            sessionid = $sessionid
            name      = $APP_NAME
            ownerid   = $OWNER_ID
        }

        $loginResponse = Invoke-RestMethod `
            -Uri $KEYAUTH_API `
            -Method Post `
            -Body $loginBody

        if ($loginResponse.success -eq $true) {
            Write-Host "Login Success" -ForegroundColor Green
            return $true
        }
        else {
            Write-Host "Login Failed" -ForegroundColor Red
            return $false
        }
    }
    catch {
        Write-Host "Connection Error" -ForegroundColor Red
        return $false
    }
}

# ================= FUNCTIONS =================

function Optimize-Network {

    Write-Host "Optimizing Network..." -ForegroundColor Cyan

    netsh int tcp set global autotuninglevel=normal
    netsh int tcp set global rss=enabled
    netsh int tcp set global chimney=enabled
    netsh int tcp set global timestamps=disabled

    ipconfig /flushdns | Out-Null
}

function Optimize-InputLag {

    Write-Host "Reducing Input Lag..." -ForegroundColor Cyan

    reg add "HKLM\SYSTEM\CurrentControlSet\Services\kbdclass\Parameters" `
    /v "KeyboardDataQueueSize" /t REG_DWORD /d 50 /f >$null

    reg add "HKLM\SYSTEM\CurrentControlSet\Services\mouclass\Parameters" `
    /v "MouseDataQueueSize" /t REG_DWORD /d 50 /f >$null

    reg add "HKCU\Control Panel\Mouse" `
    /v "MouseSpeed" /t REG_SZ /d "0" /f >$null
}

function Optimize-Visuals {

    Write-Host "Optimizing Visuals..." -ForegroundColor Cyan

    reg add "HKCU\Software\Microsoft\Windows\CurrentVersion\Explorer\VisualEffects" `
    /v "VisualFXSetting" /t REG_DWORD /d 2 /f >$null
}

function Optimize-Gaming {

    Write-Host "Boosting Gaming..." -ForegroundColor Cyan

    reg add "HKCU\Software\Microsoft\GameBar" `
    /v "AutoGameModeEnabled" `
    /t REG_DWORD /d 1 /f >$null

    reg add "HKCU\System\GameConfigStore" `
    /v "GameDVR_Enabled" `
    /t REG_DWORD /d 0 /f >$null
}

function Clean-Temp {

    Write-Host "Cleaning Temp..." -ForegroundColor Cyan

    Remove-Item "$env:TEMP\*" `
    -Force -Recurse `
    -ErrorAction SilentlyContinue

    Remove-Item "C:\Windows\Temp\*" `
    -Force -Recurse `
    -ErrorAction SilentlyContinue
}

function Restore-Defaults {

    Write-Host "Restoring Defaults..." -ForegroundColor Yellow

    reg delete "HKCU\Software\Microsoft\GameBar" `
    /v "AutoGameModeEnabled" /f 2>$null

    reg delete "HKCU\System\GameConfigStore" `
    /v "GameDVR_Enabled" /f 2>$null
}
